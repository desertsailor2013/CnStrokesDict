### 4.2 技术实现

#### 4.2.1 TTS集成
```kotlin
class TtsManager(context: Context) {
    private var tts: TextToSpeech? = null
    
    fun init() {
        tts = TextToSpeech(context) { status ->
            if (status == TextToSpeech.SUCCESS) {
                tts?.language = Locale.CHINA)
            }
        }
    }
    
    fun speak(text: String, speed: Float = 1.0f) {
        tts?.setSpeechRate(speed)
        tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, null)
    }
    
    fun stop() {
        tts?.stop()
    }
}
```

#### 4.2.2 听写逻辑
```kotlin
class DictationManager {
    private val words = mutableListOf<WordEntry>()
    private var currentIndex = 0
    private val errors = mutableListOf<WordEntry>()
    
    fun start(words: List<WordEntry>) {
        this.words.clear()
        this.words.addAll(words.shuffled())
        currentIndex = 0
        errors.clear()
    }
    
    fun next(): WordEntry? {
        if (currentIndex >= words.size) return null
        return words[currentIndex++]
    }
    
    fun markError(word: WordEntry) {
        errors.add(word)
    }
    
    fun getErrors(): List<WordEntry> = errors
    
    fun getProgress(): Pair<Int, Int> {
        return Pair(currentIndex, words.size)
    }
}
```

## 九、风险与对策

| 风险 | 影响 | 对策 |
|------|------|------|
| 联网API不稳定 | 更新失败 | 重试机制+离线缓存 |
| TTS发音不标准 | 听写体验差 | 支持多种TTS引擎 |
| 词库数据量大 | 下载慢 | 增量更新+压缩 |
| 数据库迁移复杂 | 升级困难 | 设计合理的迁移脚本 |
| 兼容性问题 | 部分设备崩溃 | 充分测试+降级方案 |
