### 2.2 技术实现

#### 2.2.1 遗忘曲线算法
```kotlin
class SpacedRepetition {
    /**
     * 计算下次复习时间
     * @param repetition 复习次数
     * @param easeFactor 难度因子
     * @param interval 间隔天数
     * @return 下次复习时间（天）
     */
    fun calculateNextReview(
        repetition: Int,
        easeFactor: Double,
        interval: Int,
    ): Double {
        return when (repetition) {
            0 -> 1.0
            1 -> 6.0
            else -> interval * easeFactor
        }
    }

    /**
     * 更新难度因子
     * @param quality 答题质量（0-5）
     * @param easeFactor 当前难度因子
     * @return 新的难度因子
     */
    fun updateEaseFactor(quality: Int, easeFactor: Double): Double {
        val newFactor = easeFactor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))
        return newFactor.coerceAtLeast(1.3)
    }
}
```

#### 2.2.2 推荐引擎
```kotlin
class RecommendationEngine(
    private val learningDao: LearningDao,
) {
    /**
     * 获取推荐复习内容
     */
    suspend fun getReviewRecommendations(limit: Int = 10): List<ReviewItem> {
        val now = System.currentTimeMillis()
        val items = mutableListOf<ReviewItem>()

        // 获取需要复习的词语
        val dueItems = learningDao.getDueReviewItems(now)
        items.addAll(dueItems)

        // 获取错误频率高的词语
        val errorItems = learningDao.getFrequentErrors(limit)
        items.addAll(errorItems)

        // 获取新词语
        val newItems = learningDao.getUnlearnedItems(limit)
        items.addAll(newItems)

        return items.distinctBy { it.word }.take(limit)
    }
}
```

#### 3.1.1 功能设计

- 支持人机对战和人人对战
- 有时间限制和提示功能

#### 3.1.2 技术实现
```kotlin
class IdiomChainGame {
    private val usedIdioms = mutableSetOf<String>()
    private var currentIdiom: String = ""

    /**
     * 开始游戏
     */
    fun startGame(): String {
        usedIdioms.clear()
        currentIdiom = getRandomIdiom()
        usedIdioms.add(currentIdiom)
        return currentIdiom
    }

    /**
     * 玩家出牌
     * @param idiom 玩家说出的成语
     * @return 游戏结果
     */
    fun play(idiom: String): GameResult {
        if (idiom !in getAllIdioms()) {
            return GameResult.Invalid("不是有效成语")
        }
        if (idiom in usedIdioms) {
            return GameResult.Invalid("成语已使用")
        }
        if (idiom[0] != currentIdiom.last()) {
            return GameResult.Invalid("开头字不匹配")
        }

        usedIdioms.add(idiom)
        currentIdiom = idiom
        return GameResult.Success(idiom)
    }

    /**
     * 获取电脑出牌
     */
    fun getComputerPlay(): String? {
        val lastChar = currentIdiom.last()
        val available = getAllIdioms().filter {
            it[0] == lastChar && it !in usedIdioms
        }
        return available.randomOrNull()
    }
}

sealed class GameResult {
    data class Success(val idiom: String) : GameResult()
    data class Invalid(val message: String) : GameResult()
}
```

## 九、风险与对策

| 风险 | 影响 | 对策 |
|------|------|------|
| AI算法不准确 | 推荐效果差 | 使用成熟算法+人工调优 |
| 游戏体验差 | 用户流失 | 优化游戏机制+用户反馈 |
| 社交功能复杂 | 开发周期长 | 分阶段实现+优先核心功能 |
| 专业术语获取困难 | 词库不完整 | 使用公开数据集+手动补充 |
| 存储空间不足 | 无法安装 | 压缩数据+按需下载 |
